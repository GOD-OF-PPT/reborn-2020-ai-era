const consumerDriven=new Set([...driven,'RelayPreparationEditor','PreparationFields','SupplierField','SelectField','RelayPreparationSchedule','RelayBusiness','RelayPeriodCard','RelayJob','ClockAdvance','RelayCommandDetails'])
function consumerElements(value:unknown,path:string):Element[]{
 if(Array.isArray(value))return value.flatMap((child,index)=>consumerElements(child,path+'/'+index))
 if(!isValidElement<Record<string,unknown>>(value))return []
 const own=[value,...consumerElements(value.props.children,path+'/children'),...consumerElements(value.props.footer,path+'/footer')]
 if(typeof value.type==='function'&&consumerDriven.has(value.type.name)){const component=value.type as (props:Record<string,unknown>)=>unknown;own.push(...consumerElements(mounted(path+':'+component.name,()=>component(value.props)),path+'/render'))}
 return own
}
function consumerDriver(name:string,render:()=>unknown){
 let tree:unknown;const all=()=>consumerElements(tree,name),refresh=()=>{tree=mounted(name,render)};refresh()
 return {refresh,all,text:()=>all().map(node=>text(node.props.children)).join(' '),
  button(label:string){const node=all().find(e=>e.type==='button'&&text(e.props.children).includes(label));if(!node)throw Error('CONSUMER_BUTTON_REQUIRED:'+label);return node},
  click(label:string){const node=all().find(e=>e.type==='button'&&text(e.props.children).includes(label));if(!node||node.props.disabled||typeof node.props.onClick!=='function')throw Error('CONSUMER_ENABLED_BUTTON_REQUIRED:'+label);node.props.onClick();refresh()},
  edit(label:string,value:string){const parent=all().find(e=>e.type==='label'&&text(e.props.children).includes(label)),node=parent&&consumerElements(parent,name+'/field').find(e=>e.type==='input'||e.type==='select');if(!node||typeof node.props.onChange!=='function'||node.props.disabled)throw Error('CONSUMER_EDITABLE_FIELD_REQUIRED:'+label);node.props.onChange({target:{value}});refresh()},
 }
}
