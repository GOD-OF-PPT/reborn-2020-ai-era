
// Additional fixed normal-prefix readers are isolated from the original
// five-save setup. They import existing bytes and execute no business command.
type NativeReaderFixture={controller:GameController;store:IndexedDbRunStore;props:()=>GamePanelProps;target:Extract<RelaySceneTarget,{kind:'period'}>;raw:string}
let nativeReader86:NativeReaderFixture|undefined,nativeReader87:NativeReaderFixture|undefined
async function originalReaderPrefix(raw:string,name:string):Promise<NativeReaderFixture>{
 const codec=createCampaignCodec(),store=new IndexedDbRunStore({codec,factory:new IDBFactory(),keyRange:IDBKeyRange,name}),controller=new GameController({codec,store,locks:new MemoryWriterAccess(),id:()=>name+'-import'})
 await controller.initialize();await controller.importNew(raw)
 const view=controller.snapshot.view!,f=features(view),business=f.relay!.businesses[0]!,station=f.relay!.stations.find(s=>s.businessId===business.id)!,period=f.relay!.periods.find(p=>p.businessId===business.id&&p.from==='2024-10-09')!,recordMemory=createRelayRecordMemory(view.runId),draftMemory=createRelayResponseDraftMemory(view.runId)
 expect(period).toBeDefined();expect(controller.exportSaved()).toBe(raw)
 const target:Extract<RelaySceneTarget,{kind:'period'}>={kind:'period',runId:view.runId,accountId:business.accountId,businessId:business.id,stationId:station.id,periodId:period.id}
 const props=():GamePanelProps=>{const current=controller.snapshot.view!;return {controller,view:current,features:features(current),accountId:business.accountId,request:vi.fn<GamePanelProps['request']>(),textOnly:true,relayRecordMemory:recordMemory,relayResponseDraftMemory:draftMemory}}
 return {controller,store,props,target,raw}
}
beforeAll(async()=>{nativeReader86=await originalReaderPrefix(source86,'reader-native-technical86');nativeReader87=await originalReaderPrefix(source87,'reader-native-technical87')},20000)
afterAll(async()=>{for(const fixture of [nativeReader86,nativeReader87])if(fixture)await fixture.controller.close()})
const ticketRadios=(ui:ReturnType<typeof consumerDriver>)=>ui.all().filter(e=>e.type==='label'&&consumerElements(e,'native-ticket-label').some(child=>child.type==='input'&&child.props.type==='radio'))

 const fixture=nativeReader86!,p=fixture.props(),target=fixture.target,relay=p.features.relay!,model=relayRecordRows(p,target),due=model.rows.find(row=>row.kind==='receivable'&&row.periodId===target.periodId&&row.attention==='due')!,request=vi.fn<GamePanelProps['request']>(),raw=fixture.controller.exportSaved(),ui=consumerDriver('native-parent86-due-first',()=>RelayRecordBook({p:{...p,request},target}))
 expect(model.reason).toBeNull();expect(p.view.revision).toBe(86);expect(due).toBeDefined();expect(due.key).toBe('receivable:'+due.receivableId)

 expect(ui.all().find(e=>e.type==='h3')?.props.children).toBe('这张原单，这次先收多少？')

 expect(ui.text()).toContain(due.receivableId!);expect(relay.receivables.find(i=>i.id===due.receivableId)).toMatchObject({dueAt:'2024-11-07',amountFen:1216,collectedFen:0,reversedFen:0})
 ui.click('先收起本次填写，仍留这张原票');expect(ui.text()).toContain(due.receivableId!)

 ui.click('核对这张原单，本次先收多少');expect(ui.all().find(e=>e.type==='input'&&e.props.inputMode==='decimal')?.props.value).toBe('')
 expect(request).not.toHaveBeenCalled();expect(fixture.controller.exportSaved()).toBe(raw);expect(raw).toBe(fixture.raw);expect(fixture.controller.snapshot.view).toBe(p.view)
})
